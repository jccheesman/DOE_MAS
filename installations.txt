# Ollama setup (on Jetstream VM):
# 1. Install Ollama: curl -fsSL https://ollama.com/install.sh | sh
# 2. Pull the model: ollama pull llama3.1:70b
# 3. Start server (bind to all interfaces for remote access):
#    OLLAMA_HOST=0.0.0.0:11434 ollama serve
#
# On the client machine (where this code runs), set the environment variable:
# export OLLAMA_API_BASE=http://<jetstream-ip>:11434
#
